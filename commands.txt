2. DOCKER COMPOSE (DIA A DIA)
----------------------------------------------------------------
docker compose ps                         # estado dos serviços
docker compose config > /dev/null && echo "YAML ok"

docker compose up -d --build              # subir/atualizar tudo
docker compose up -d --build backend      # só um serviço
docker compose restart backend            # reinicia (NÃO relê o env_file)
docker compose up -d --force-recreate backend   # recria e relê backend.env
docker compose stop                       # para tudo (mantém os dados)
docker compose start                      # volta a rodar

# Logs
docker compose logs -f                    # todos
docker compose logs -f backend            # um serviço
docker compose logs --tail=100 backend adk-service
docker compose logs --tail=50 caddy

# Entrar num container
docker compose exec backend sh
docker compose exec postgres psql -U app -d stakeholder   # troque app/stakeholder pelo seu .env

4. BANCO DE DADOS (PRISMA)
----------------------------------------------------------------
# Aplicar migrations (se existir backend/prisma/migrations)
docker compose exec backend npm run prisma:migrate

# Criar tabelas direto do schema (banco novo, sem migrations)
docker compose exec backend npx prisma db push

# Banco criado com "db push" (antes de as migrations irem pro git): o
# "npm run prisma:migrate" marca a 20261002202542_init como aplicada sozinho
# (backend/prisma/baseline.js) e depois aplica o resto. Nada manual.
# Conferir (só leitura) o que falta no banco em relação ao schema:
docker compose run --rm -T backend sh -c 'npx prisma migrate diff --from-url "$DATABASE_URL" --to-schema-datamodel prisma/schema.prisma --script'

# Backup antes de um deploy com migration (troque app/stakeholder pelo seu .env):
docker compose exec -T postgres pg_dump -U app -d stakeholder > backup_$(date +%F_%H%M).sql

# Seed (cria o superadmin e dados iniciais)
docker compose exec backend npm run seed

# Listar tabelas
docker compose exec postgres psql -U app -d stakeholder -c '\dt'

# Gerar a migration inicial 0_init (uma vez só, depois commitar a pasta)
mkdir -p backend/prisma/migrations/0_init
docker compose exec -T backend npx prisma migrate diff \
  --from-empty --to-schema-datamodel prisma/schema.prisma --script \
  > backend/prisma/migrations/0_init/migration.sql
printf 'provider = "postgresql"\n' > backend/prisma/migrations/migration_lock.toml
docker compose exec backend npx prisma migrate resolve --applied 0_init

5. SENHA DO SUPERADMIN
----------------------------------------------------------------
# Antes, confira o nome do campo da senha e o custo do bcrypt:
grep -n -A15 "model User" backend/prisma/schema.prisma
grep -n "bcrypt\|password" backend/src/seed.ts

# Troca (ajuste "passwordHash" se o campo tiver outro nome).
# A senha é digitada sem aparecer na tela e não vai para o histórico.
read -rs -p "Nova senha: " NEW_PW; echo
docker compose exec -e NEW_PW="$NEW_PW" backend node -e '
const bcrypt = require("bcryptjs");
const { PrismaClient } = require("@prisma/client");
(async () => {
  const prisma = new PrismaClient();
  const hash = await bcrypt.hash(process.env.NEW_PW, 10);
  const r = await prisma.user.updateMany({
    where: { email: "superadmin@gmail.com" },
    data: { passwordHash: hash },
  });
  console.log("atualizados:", r.count);
  await prisma.$disconnect();
})();
'
unset NEW_PW

# Atenção: se o upsert do seed sobrescreve a senha no "update",
# rodar "npm run seed" de novo volta a senha padrão.

6. BACKUP E RESTAURAÇÃO
----------------------------------------------------------------
mkdir -p ~/backups

# Banco
docker compose exec -T postgres pg_dump -U app stakeholder \
  | gzip > ~/backups/db-$(date +%F).sql.gz

# PDFs (volume "uploads"; o prefixo é o nome da pasta do projeto em minúsculas)
docker run --rm -v stackholderteste_uploads:/data -v ~/backups:/backup alpine \
  tar czf /backup/uploads-$(date +%F).tar.gz -C /data .

# Copiar para o seu PC (rode no PC)
scp deploy@IP_DA_VPS:~/backups/*.gz .

# Restaurar o banco (apaga o conteúdo atual do banco!)
gunzip -c ~/backups/db-AAAA-MM-DD.sql.gz | docker compose exec -T postgres psql -U app -d stakeholder

# Restaurar os PDFs
docker run --rm -v stackholderteste_uploads:/data -v ~/backups:/backup alpine \
  tar xzf /backup/uploads-AAAA-MM-DD.tar.gz -C /data
